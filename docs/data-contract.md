# data.json — состав полей

Файл `~/.claude/usage-report/data.json` пересобирается скриптом при каждом запуске; тот же объект
вставляется в шаблон как `const DATA = …`. Время — местное (`tz_offset_hours` из настроек),
даты — строки `ГГГГ-ММ-ДД`, секунды — целые, токены — целые.

```jsonc
{
  "generated": "2026-09-12 04:45",      // когда собрано
  "today": "2026-09-12",
  "groups": [{"id": "core", "label": "Основная работа"}, …],   // порядок = порядок цветов
  "work_groups": ["core", "work"],       // какие группы считаются «работой»
  "projects": {"GW-all-ideas": "core", "Astroclub-virtual": "personal", …},   // проект → группа
  "model_labels": {"claude-opus-5": "Opus 5", …},
  "months": ["2026-09", "2026-08", …],   // месяцы, где есть данные, новые первыми

  "days": [                              // по возрастанию даты; только дни с активностью
    {"d": "2026-09-11",
     "p": {                              // проект →
       "GW-all-ideas": {
         "sec": 9720,                    // активные секунды (промежутки ≤ idle_minutes)
         "hours": [0, 0, …, 1800, …],    // те же секунды по часам суток 0–23
         "msgs": 42,                     // сообщений пользователя (без результатов инструментов)
         "sessions": ["b622af0b-…", …],  // id сессий, живших в этот день
         "ins": {                        // куда уходит расход, в единицах нагрузки (токены × веса)
           "units": 12.4, "ctx": {"lt50": 0.1, "50_100": 2.0, "100_150": 3.1, "gt150": 7.2},   // по размеру контекста вызова
           "effort": {"high": 200, "xhigh": 60},          // вызовов по уровню усилия
           "side_units": 0.2,                             // из них субагенты
           "cache_create_units": 2.6,                     // запись в кэш
           "skills": {"claude-api": 0.4},                 // перечитывание текста навыка из кэша
           "sessions": 3, "sessions_short": 1, "first_call_units": 0.8   // сессии, начатые в этот день; короче 3 вызовов; стоимость первых вызовов
         },
         "models": {                     // модель →
           "claude-opus-5": {"calls": 286, "input": 6000, "cache_create": 1600000,
                             "cache_read": 83853000, "output": 440000, "thinking": 102000}
         }}}}
  ],

  "weeks": [                             // окна недельного лимита (от week_reset), по возрастанию
    {"start": "2026-09-10", "end": "2026-09-16", "start_at": "2026-09-10 20:00",
     "plan": "max5", "plan_label": "Max 5x",
     "units": 105.5,                     // единицы нагрузки = токены × веса, все модели
     "budget": 527.6,                    // бюджет окна с учётом акции; null, если тариф не откалиброван
     "pct": 20.0,                        // units / budget × 100; null без бюджета
     "fable_units": 83.9, "fable_budget": 246.6, "fable_pct": 34.0,   // отдельная шкала Fable
     "boost": 1.5,                       // множитель акции в это окно
     "estimated": false,                 // true = у тарифа нет своего показания (процентов нет)
     "partial": true}                    // окно ещё идёт
  ],

  "commits": {                         // день → проект → [24] коммитов по часам (локальный git, авторы из настроек)
    "2026-09-11": {"GW-all-ideas": [0, 0, …, 3, …]}
  },
  "git": {"repos": 66, "authors": ["thinkpload@gmail.com", …]},

  "limits": {
    "calibrated": true,                  // есть ли показание для текущего тарифа
    "plan_now": "Max 5x",
    "budgets": {"Max 5x": {"all": 351.6, "fable": 164.6, "estimated": false}, "Pro": {…, "estimated": true}},
    "observations": [{"at": "2026-09-12 05:50", "pct": 20, "fable_pct": 34}],
    "week_reset": "2026-09-17 20:00"     // или null (недели с понедельника)
  },
  "idle_minutes": 10,
  "config_path": "C:\\Users\\…\\config.json"
}
```

Что считать самому из `days`: часы проекта/группы за период — сумма `sec`; вызовы и токены —
сумма по `models`; сессии за период — объединение `sessions` (одна сессия может жить несколько дней);
тепловая карта «день недели × час» — сумма `hours` по дню недели даты `d`; «размышления» —
`thinking` входит в `output`, не складывать их.
